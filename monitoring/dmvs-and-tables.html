❓ "Is replication healthy right now?" — the 60-second sweep

USE distribution;

-- 1. rollup per publication (status 6=failed, 5=retrying are your fires)
EXEC sp_replmonitorhelppublication @publisher = N'PUB01';

-- 2. per subscription with latency
EXEC sp_replmonitorhelpsubscription @publisher = N'PUB01', @publication_type = 0;

-- 3. any recent agent errors?
SELECT TOP 20 time, error_text = CONVERT(nvarchar(4000), error_text), source_name
FROM dbo.MSrepl_errors ORDER BY time DESC;

-- 4. are REPL jobs running / recently failed?
SELECT j.name, c.name AS category, a.start_execution_date, a.stop_execution_date
FROM msdb.dbo.sysjobs j
JOIN msdb.dbo.syscategories c ON c.category_id = j.category_id
LEFT JOIN msdb.dbo.sysjobactivity a ON a.job_id = j.job_id
  AND a.session_id = (SELECT MAX(session_id) FROM msdb.dbo.syssessions)
WHERE c.name LIKE 'REPL-%'
ORDER BY c.name, j.name;

❓ "How big is the backlog?"

USE distribution;

-- raw queue depth (fast, approximate)
SELECT COUNT(*) AS commands_in_queue FROM dbo.MSrepl_commands WITH (NOLOCK);

-- per subscription, with estimated apply time (what the GUI shows)
EXEC sp_replmonitorsubscriptionpendingcmds
     @publisher = N'PUB01', @publisher_db = N'SalesDB', @publication = N'SalesPub',
     @subscriber = N'SUB01', @subscriber_db = N'SalesReporting', @subscription_type = 0;

-- biggest transactions in the queue (large-transaction incidents)
SELECT TOP 10 t.xact_seqno, t.entry_time, cmds = COUNT(*)
FROM dbo.MSrepl_transactions t
JOIN dbo.MSrepl_commands c ON c.xact_seqno = t.xact_seqno
GROUP BY t.xact_seqno, t.entry_time
ORDER BY cmds DESC;

❓ "Is the Log Reader keeping up?" — publisher side

-- throughput + latency estimate per published DB
EXEC sp_replcounters;
-- columns: replicated transactions (waiting), replication rate (txn/s),
--          replication latency (s), replbeginlsn / replnextlsn

-- what is pinning the log?
SELECT name, log_reuse_wait_desc FROM sys.databases WHERE name = N'SalesDB';
-- REPLICATION here + growing log = log reader behind or CDC/repl orphaned

-- transactions committed but not yet harvested
DBCC OPENTRAN (N'SalesDB');          -- shows oldest replicated txn
EXEC sp_repltrans;                    -- list of all waiting txns (publication DB context)
🛑 Orphaned "REPLICATION" log_reuse_wait

If replication was removed carelessly (or CDC shares the log reader), the log can stay pinned forever. After confirming no publications remain and none are wanted, the documented fix is EXEC sp_removedbreplication N'SalesDB'; — and for a truly stuck log the last resort sp_repldone reset (see common errors for the exact sequence and its dangers).

❓ "What exactly is the agent doing / saying?"

USE distribution;

-- distribution agent sessions, newest first
SELECT TOP 30 a.name, h.start_time, h.duration,
       h.delivered_commands, h.current_delivery_rate, h.comments
FROM dbo.MSdistribution_history h
JOIN dbo.MSdistribution_agents a ON a.id = h.agent_id
ORDER BY h.start_time DESC;

-- log reader sessions
SELECT TOP 30 a.name, h.start_time, h.delivered_commands, h.comments
FROM dbo.MSlogreader_history h
JOIN dbo.MSlogreader_agents a ON a.id = h.agent_id
ORDER BY h.start_time DESC;

-- same pattern: MSsnapshot_history, MSmerge_history

For deep debugging, add verbose output to the agent job step and re-run: -Output C:\Temp\distrib.log -OutputVerboseLevel 2 — the full command stream and timings land in the file.

❓ "Are publisher and subscriber actually identical?" — validation

-- transactional: queue validation requests for all articles
USE SalesDB;
EXEC sp_publication_validation
     @publication = N'SalesPub',
     @rowcount_only = 2,     -- 1 = rowcount only, 2 = rowcount + binary checksum
     @full_or_fast = 2;      -- fast rowcount from metadata where possible

-- results arrive as messages in the Distribution Agent history;
-- failures also fire the 'Peer-to-peer/validation failure' alert class (20574)

-- merge equivalent
EXEC sp_validatemergesubscription
     @publication = N'FieldSalesPub',
     @subscriber = N'DEV-LAPTOP-07', @subscriber_db = N'FieldSales',
     @level = 3;             -- rowcount + checksum
